+        return _cell(self.raw, row[0], row[1], row[0] + offset, width, self.budget)

+    def zero(

+                self.integer(row, position, width) == 0,

+        self, tag: int, offset: int, count: int, *, record: bool = False

+            "A slice does not start at its ownership cursor.",

+    def value(
+        self,
+        descriptor: layout.Descriptor,
+        row: tuple[int, int],
+        offset: int,
+        previous: Mapping[str, LogicalValue],
+        depth: int,
+        owner: int,
+        name: str,
+    ) -> LogicalValue:
+        self.budget.depth(depth)
+        match descriptor:
+            case layout.Option(payload=payload):
+                tag = self.integer(row, offset)
+                _require(tag in (0, 1), "logical_shape", "Invalid option tag.")
+                if tag == 0:
+                    self.zero(row, offset + 8, payload)
+                    return None
+                return self.value(
+                    payload, row, offset + 8, previous, depth + 1, owner, name
+                )
+            case layout.Variant(nullable=nullable, target_tags=tags):
+                tag = self.integer(row, offset)
+                _require(
+                    (0 if nullable else 1) <= tag <= len(tags),
+                    "logical_shape",
+                    "Invalid variant tag.",
+                )

+                    if i != tag:

+                            row, offset + i * 8, layout.TableSlice(target, "record")
+                        )
+                if tag == 0:
+                    return None
+                return self.value(
+                    layout.TableSlice(tags[tag - 1], "record"),
+                    row,
+                    offset + tag * 8,
+                    previous,
+                    depth + 1,
+                    owner,
+                    name,
+                )
+            case layout.TableSlice(target_tag=target, mode=mode):
+                begin = self.integer(row, offset, 4)
+                count = self.integer(row, offset + 4, 4)
+                self.count_limit(owner, name, count)
+                claimed = self.claim(target, begin, count, record=mode == "record")
+                if mode == "record":

+                    self.decode_row(target, index, depth + 1) for index in claimed
+                )
+            case layout.TextSlice(kind=kind, literals=literals):
+                begin = self.integer(row, offset, 4)
+                count = self.integer(row, offset + 4, 4)
+                self.claim(1, begin, count)

+                    kind != "witness" or 1 <= count <= 15,
+                    "logical_shape",
+                    "Invalid instruction witness length.",

+                    kind != "digest" or count == 64,
+                    "logical_shape",
+                    "Invalid digest length.",
+                )

+                raw = self.raw[start : start + count]

+                    self.budget.scan(count + count * 2)

+                copied = bytes(raw)
+                self.budget.scan(count)
+                try:
+                    text = copied.decode("utf-8", errors="strict")
+                except UnicodeDecodeError:
+                    _refuse("logical_shape", "Invalid UTF-8 text.")
+                if kind == "digest":
+                    self.budget.scan(count)
+                    _require(
+                        all(c in "0123456789abcdef" for c in text),
+                        "logical_shape",
+                        "Invalid digest text.",
+                    )
+                if kind == "literal":
+                    self.budget.scan(count * len(literals))
+                    _require(text in literals, "logical_shape", "Invalid literal text.")
+                return text
+            case layout.Signed64():
+                sign, magnitude = (
+                    self.integer(row, offset),
+                    self.integer(row, offset + 8),
+                )
+                _require(
+                    sign in (0, 1)
+                    and magnitude <= (1 << 63) - (1 - sign)
+                    and not (sign == 1 and magnitude == 0),
+                    "logical_shape",
+                    "Invalid signed value.",
+                )
+                return -magnitude if sign else magnitude
+            case (
+                layout.UInt64()
+                | layout.Boolean64()
+                | layout.ExactNumber()
+                | layout.EndFrom()
+            ):
+                value = self.integer(row, offset)
+                if isinstance(descriptor, layout.Boolean64):
+                    _require(value in (0, 1), "logical_shape", "Invalid Boolean value.")
+                    return bool(value)
+                if isinstance(descriptor, layout.ExactNumber):
+                    _require(
+                        value == descriptor.value,
+                        "logical_shape",
+                        "Invalid fixed number.",
+                    )
+                if isinstance(descriptor, layout.EndFrom):
+                    value += cast(int, previous[descriptor.prior_name])

+                        value <= 1 << 64,
+                        "logical_shape",
+                        "An endpoint exceeds unsigned range.",
+                    )
+                return value
+        raise TypeError("Invalid installed descriptor.")

+    def decode_row(self, tag: int, index: int, depth: int) -> LogicalValue:

+def _check_retained_limits(
+    root: LogicalObject, lower: _Lower, budget: WorkBudget
+) -> None:
+    original = _get(root, "original", budget)
+    size = cast(int, _get(original, "bytes", budget))
+    budget.scalar()
+    _require(
+        size <= lower.transport["original_bytes"],
+        "input_limit",
+        "The original byte cap is exceeded.",

+    guards = cast(tuple[LogicalValue, ...], _get(scope, "guard_instructions", budget))
+    guard_map: dict[int, tuple[int, int]] = {}
+    for guard in guards:
+        address = cast(int, _get(guard, "address", budget))
+        owner = cast(int, _get(guard, "owner", budget))
+        witness = cast(str, _get(guard, "witness_hex", budget))

+            address not in guard_map, "logical_shape", "A guard address is ambiguous."
+        )

+    routes = cast(tuple[LogicalValue, ...], _get(scope, "routes", budget))

+        path = cast(tuple[int, ...], _get(route, "instruction_path", budget))

+                cursor < len(path) and path[cursor] == entry,
+                "logical_shape",
+                "A route owner group does not start at its entry.",
+            )
+            count = byte_count = last_size = 0
+            while cursor < len(path):
+                budget.scalar()
+                guard = guard_map.get(path[cursor])
+                _require(
+                    guard is not None, "logical_shape", "A route position has no guard."
+                )
+                assert guard is not None
+                if guard[0] != entry:
+                    break
+                count += 1
+                last_size = guard[1]
+                byte_count += last_size
+                cursor += 1
+                _require(
+                    count - 1 <= lower.capture["thunk_neutral_instructions"],
+                    "input_limit",
+                    "The neutral instruction cap is exceeded.",
+                )
+            _require(count > 0, "logical_shape", "A route owner group is empty.")
+            _require(
+                byte_count - last_size <= lower.capture["thunk_neutral_bytes"],
+                "input_limit",
+                "The neutral byte cap is exceeded.",
+            )
+        _require(
+            cursor == len(path), "logical_shape", "A route path has no owner group."
+        )

+def _canonical_bytes(root: LogicalObject, lower: _Lower, budget: WorkBudget) -> bytes:
+    output = bytearray()
+    active: list[tuple[int, int]] = []
+    cap = min(budget.limits.logical_data_bytes, lower.transport["data_bytes"])
+
+    def append(token: str) -> None:
+        count = len(token)
+        _require(
+            count + 1 <= cap - len(output),
+            "record_limit",
+            "The logical byte cap is exceeded.",
+        )
+        for begin, maximum in active:
+            _require(
+                len(output) - begin + count + 1 <= maximum,
+                "record_limit",
+                "A subtree byte cap is exceeded.",
+            )
+        budget.scan(count * 2)
+        output.extend(token.encode("ascii"))
+
+    def string(value: str) -> None:
+        append('"')
+        for character in value:
+            budget.scan(16)
+            append(json.dumps(character, ensure_ascii=True)[1:-1])
+        append('"')
+
+    def visit(value: LogicalValue, depth: int, subtree: int | None = None) -> None:
+        budget.depth(depth)
+        budget.scalar()
+        if subtree is not None:
+            active.append((len(output), subtree))
+        if type(value) is LogicalObject:
+            append("{")
+            for i, (key, child) in enumerate(value.entries):
+                if i:
+                    append(",")
+                string(key)
+                append(":")
+                bound = None
+                if value is root and key in ("full_plan", "scope"):
+                    bound = lower.capture[
+                        "full_plan_bytes" if key == "full_plan" else "scope_bytes"
+                    ]
+                visit(child, depth + 1, bound)
+            append("}")
+        elif type(value) is tuple:
+            append("[")
+            for i, child in enumerate(value):
+                if i:
+                    append(",")
+                visit(child, depth + 1)
+            append("]")
+        elif type(value) is str:
+            string(value)
+        elif value is None:
+            append("null")
+        elif type(value) is bool:
+            append("true" if value else "false")
+        elif type(value) is int:
+            budget.scan(21)
+            append(str(value))
+        else:
+            raise TypeError("Invalid internal logical value.")
+        if subtree is not None:
+            active.pop()
+
+    visit(root, 1)
+    _require(len(output) < cap, "record_limit", "No space remains for the final LF.")
+    budget.scan(1)
+    output.append(10)
+    budget.scan(len(output))
+    return bytes(output)
+

+def _compare_complete(
+    actual: bytes,
+    expected: ExpectedCollector,
+    directory: DirectoryView,
+    budget: WorkBudget,
+) -> str:
+    budget.scan(max(len(actual), len(expected.logical_raw)))
+    _require(
+        actual == expected.logical_raw,
+        "logical_mismatch",
+        "The complete logical bytes differ.",
+    )
+    budget.scan(len(actual))
+    digest = sha256(actual).digest()
+    budget.scan(32)
+    _require(
+        digest == directory.data_digest,
+        "logical_mismatch",
+        "The directory data digest differs.",
+    )
+    for text, stored in (
+        (expected.renderer_sha256, directory.renderer_digest),
+        (expected.native_source_sha256, directory.native_digest),
+    ):
+        budget.scan(64 + 32 + 32)
+        _require(
+            bytes.fromhex(text) == stored,
+            "source_identity",
+            "A directory source identity differs.",
+        )
+    budget.scan(32 + 64)
+    return digest.hex()

+class Cells:
+    def __init__(self) -> None:
+        self.rows: dict[int, list[bytearray]] = {tag: [] for tag in range(1, 48)}
+        self.blob = bytearray()
+
+    def reserve(self, tag: int, values: list[Any]) -> tuple[int, int]:
+        begin = len(self.rows[tag])
+        self.rows[tag].extend(bytearray(SIZES[tag]) for _ in values)
+        for index, value in enumerate(values, begin):
+            self.row(tag, index, value)
+        return begin, len(values)
+
+    def row(self, tag: int, index: int, value: Any) -> None:
+        entries = SPECS[tag]
+        if KINDS[tag] == "scalar":
+            value = {"value": value}
+        elif KINDS[tag] == "pair":
+            value = dict(zip((f[1] for f in entries), value, strict=True))
+        for _, name, _, offset, descriptor in entries:
+            self.cell(self.rows[tag][index], offset, descriptor, value[name], value)
+
+    def cell(
+        self, row: bytearray, offset: int, desc: str, value: Any, previous: Any
+    ) -> None:
+        if desc.startswith("o:"):
+            pack_into("<Q", row, offset, int(value is not None))
+            if value is not None:
+                self.cell(row, offset + 8, desc[2:], value, previous)
+        elif desc.startswith(("v:", "v0:")):
+            choices = desc.split(":", 1)[1].split(",")
+            if value is None:
+                return
+            name = value.get(
+                "kind",
+                "exact_signal" if "signal" in value else "exit_from_supported_bits",
+            )
+            if name in ("direct_call", "indirect_call"):
+                name = "call"
+            if name == "indirect_jump":
+                name = "jump"
+            index = choices.index(name)
+            pack_into("<Q", row, offset, index + 1)
+            self.cell(row, offset + 8 * (index + 1), "r:" + name, value, previous)
+        elif desc.startswith(("a:", "r:")):
+            target = NAMES[desc[2:]]
+            begin, count = self.reserve(
+                target, value if desc.startswith("a:") else [value]
+            )
+            pack_into("<II", row, offset, begin, count)
+        elif desc in ("s", "h", "w") or desc.startswith("e:"):
+            raw = bytes.fromhex(value) if desc == "w" else value.encode("utf-8")
+            pack_into("<II", row, offset, len(self.blob), len(raw))
+            self.blob.extend(raw)
+        elif desc == "i":
+            pack_into("<QQ", row, offset, int(value < 0), abs(value))
+        else:
+            if desc.startswith("end:"):
+                value -= previous[desc[4:]]
+            pack_into("<Q", row, offset, value)
+
+
+def built(
+    record: Any = None, machine: elf.Machine = 62, placement: int = 0
+) -> tuple[Fixture, compiled.ExpectedCollector]:
+    record = sample() if record is None else record
+    cells = Cells()
+    cells.reserve(47, [record])
+    f = fixture(machine)
+    f.data[0x2000 : 0x2000 + 40168] = witness(machine)
+    expected_raw = (
+        json.dumps(record, sort_keys=True, ensure_ascii=True, separators=(",", ":"))
+        + "\n"
+    ).encode("ascii")
+    f.data[0x20000:0x20060] = (
+        sha256(expected_raw).digest() + b"\x22" * 32 + b"\x33" * 32
+    )
+    cursor = 0xBD00 + placement
+    offsets = []
+    for tag in range(1, 48):
+        cursor = (cursor + 7) // 8 * 8
+        raw = bytes(cells.blob) if tag == 1 else b"".join(cells.rows[tag])
+        count = len(cells.blob) if tag == 1 else len(cells.rows[tag])
+        storage = raw if count else bytes(SIZES[tag])
+        assert cursor + len(storage) <= 0x10000
+        f.data[cursor : cursor + len(storage)] = storage
+        f.put(
+            0x20060 + (tag - 1) * 56,
+            "6Q",
+            tag,
+            count,
+            SIZES[tag],
+            count * SIZES[tag],
+            SIZES[tag],
+            len(storage),
+        )
+        name = _tables()[tag - 1][1]
+        f.put(f.local_symbols[name] + 8, "QQ", BASE + cursor, len(storage))
+        f.put(0x1000 + (tag - 1) * 24 + 16, "q", BASE + cursor)
+        offsets.append(cursor)
+        cursor += len(storage)
+    f.table_offsets = tuple(offsets)
+    return f, compiled.ExpectedCollector(expected_raw, "22" * 32, "33" * 32)
+

+    expected: compiled.ExpectedCollector,

+        bytes(f.data),

+        profile=cast(layout.ProfileId, PROFILES[f.machine]),

+def thaw(value: compiled.LogicalValue) -> Any:
+    if type(value) is compiled.LogicalObject:
+        return {key: thaw(child) for key, child in value.entries}
+    if type(value) is tuple:
+        return [thaw(child) for child in value]
+    return value

+def private_rows(tag: int, value: Any) -> compiled._Decoder:
+    cells = Cells()
+    cells.reserve(tag, [value])
+    raw = bytearray()
+    tables = []
+    for number, name, _, size, _ in _tables():
+        content = bytes(cells.blob) if number == 1 else b"".join(cells.rows[number])
+        count = len(content) // size
+        storage = content or bytes(size)
+        offset = len(raw)
+        raw.extend(storage)
+        tables.append(
+            compiled.TableSpan(
+                number,
+                name,
+                count,
+                size,
+                count * size,
+                size,
+                len(storage),
+                0,
+                0,
+                elf.ElfVa(offset),
+                elf.FileOffset(offset),
+                elf.Rela(0, 0, elf.FileOffset(0), elf.ElfVa(0), 8, 0, offset),
+            )
+        )
+    export = elf.Export("private", 0, 0, 0, elf.ElfVa(0), elf.FileOffset(0), 0, 0, 1, 0)
+    directory = compiled.DirectoryView(
+        export, bytes(32), bytes(32), bytes(32), tuple(tables)
+    )
+    lower = compiled._Lower(
+        *(
+            dict(zip((f[1] for f in SPECS[t]), MAXIMA[t], strict=True))
+            for t in (11, 12, 13)
+        )
+    )
+    decoder = compiled._Decoder(
+        memoryview(bytes(raw)),
+        layout._definition(PROFILES[62]),
+        directory,
+        compiled.WorkBudget(elf.InspectionLimits()),
+        compiled.CursorState(),
+        lower,
+    )
+    decoder.claim(tag, 0, 1, record=True)
+    return decoder
+

+    actual = decoder.decode_row(tag, 0, 1)
+    assert [t.count for t in decoder.directory.tables] == decoder.state.cursors
+    return thaw(actual)
